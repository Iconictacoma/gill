cmake_minimum_required(VERSION 3.20)
project(g2m VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(g2m src/main.cpp src/lexer.cpp src/parser.cpp src/ir.cpp)
target_include_directories(g2m PRIVATE include)
if(MSVC)
 target_compile_options(g2m PRIVATE /W4 /permissive-)
else()
 target_compile_options(g2m PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME g2m_hello COMMAND g2m ${CMAKE_CURRENT_SOURCE_DIR}/examples/hello.gill)
add_test(NAME g2m_no_module COMMAND g2m ${CMAKE_CURRENT_SOURCE_DIR}/examples/no_module.gill -c ${CMAKE_CURRENT_BINARY_DIR}/no_module_test)
add_test(NAME g2m_primitive_types COMMAND g2m ${CMAKE_CURRENT_SOURCE_DIR}/examples/primitive_types.gill -c ${CMAKE_CURRENT_BINARY_DIR}/primitive_types_test)

add_test(NAME g2m_main_i32 COMMAND g2m ${CMAKE_CURRENT_SOURCE_DIR}/examples/main_i32.gill -c ${CMAKE_CURRENT_BINARY_DIR}/main_i32_test)
add_test(NAME g2m_main_int COMMAND g2m ${CMAKE_CURRENT_SOURCE_DIR}/examples/main_int.gill -c ${CMAKE_CURRENT_BINARY_DIR}/main_int_test)
